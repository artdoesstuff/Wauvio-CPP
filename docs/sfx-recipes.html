SFX engine (wauvio/sfx)

Recipes, trajectories and modulators

See it in action: Build a sound from scratch

Common utilities (sfx_common.hpp)

  • sfx::Error — exception type (std::runtime_error) used for every validation failure.
  • Math: clampd, lerp, midi_to_hz, hz_to_midi, db_to_lin, lin_to_db, semis_to_ratio, is_finite.
  • Hashing / randomness: hash64, mix_seed(a, b), hash01, value_noise(x, seed), and the Rng class (next, uniform, bipolar, range, chance, index, gaussian).
  • Music tables: scale_names(), scale_degrees(i), scale_step(deg, k), chord_names(), chord_intervals(i). Scales: chromatic, major, natural_minor, harmonic_minor, pentatonic_major, pentatonic_minor, whole_tone, dorian, phrygian, lydian, mixolydian. Chords: major, minor, sus2, sus4, major7, minor7, dominant7, diminished, augmented, add9, cluster.

Traj — parameter trajectory

A piecewise function of normalised time u ∈ [0, 1] (the position inside a layer). Construct from a constant (Traj(5.0) converts implicitly) or add keys in time order with to(t, value, Interp); the interpolation of a key describes how the segment arriving at it is shaped. Interp = Linear, Exp, Smooth, Step (Exp requires both end values positive, otherwise it falls back to linear). Non-finite numbers throw.

Mod — stateless LFO

Mod::Shape = Sine, Triangle, Square, Saw, Smooth, Steps (the last two are seeded random). Fields: rate_hz, beats (if > 0 one cycle per that many beats, tempo-synced), depth (output is ±depth), phase (cycles). active(), eval(t_sec, bpm, seed). Because it is evaluated from time alone, layers can render in any order.

Helpers (sfx_helpers.hpp)

  • random_stairs(rng, n, lo, hi, snap), env_bell(center, width, keys, floor), env_swell(peak_at, start, end), env_hit(attack, floor), lfo(shape, rate_hz, depth, phase), lfo_beats(shape, beats_per_cycle, depth, phase), hashed_pattern(seed, n, density).

Recipe

Recipe is the plain-data description of an effect: name, duration (nominal seconds), tail (extra seconds), bpm (120), reverse_all, fade_in_ms (1), fade_out_ms (6) and layers. add(name) appends and returns a LayerSpec&; total_seconds().

LayerSpec

One layer = one source → envelope → processor chain → post automation → level/pan. Every builder returns *this so calls chain. Fields env, post, level, start, length (<0 = to the end), tail, pan, width (0 mono-panned … 1 independently rendered L/R), fade_in_ms, fade_out_ms, seed_offset.

inline uint64_t mix_seed(uint64_t a, uint64_t b) noexcept { return hash64(a ^ (hash64(b) + 0x9E3779B97F4A7C15ull + (a << 6) + (a >> 2))); }
inline double hash01(uint64_t x) noexcept { return static_cast<double>(hash64(x) >> 11) * (1.0 / 9007199254740992.0); }

class Rng
public:
    explicit Rng(uint64_t seed = 1) noexcept { reseed(seed); }
    void reseed(uint64_t seed) noexcept { uint64_t z = seed; for (auto& s : s_) { z += 0x9E3779B97F4A7C15ull; s = hash64(z); } }
    uint64_t next() noexcept
    double uniform() noexcept { return static_cast<double>(next() >> 11) * (1.0 / 9007199254740992.0); }   // [0,1)
    double bipolar() noexcept { return uniform() * 2.0 - 1.0; }                                              // [-1,1)
    double range(double a, double b) noexcept { return a + (b - a) * uniform(); }
    bool   chance(double p) noexcept { return uniform() < p; }
    int    index(int n) noexcept { return n <= 1 ? 0 : static_cast<int>(uniform() * n) % n; }
    double gaussian() noexcept { const double u1 = std::max(uniform(), 1e-12), u2 = uniform(); return std::sqrt(-2.0 * std::log(u1)) * std::cos(TWO_PI * u2); }

wauvio::sfx::class Traj

/// A parameter trajectory over normalised time u in [0, 1] (u = position inside the layer).
/// Keys are added in time order; each key records how the segment *arriving* at it is
/// interpolated. Exp interpolation needs both end values to be positive and falls back to
/// linear otherwise. Non-finite numbers are rejected when building, and at() sanitises its input.
///
///   Traj::exp(200, 12000)                                           // exponential sweep (musically even)
///   Traj(0).to(0.1, 1, Interp::Smooth).to(1, 0.001, Interp::Exp)    // attack then decay
class Traj
public:
    Traj(double constant = 0.0) { check(0.0, constant); k_.push_back({0.0, constant, Interp::Linear}); }   // NOLINT implicit by design
    static Traj linear(double a, double b) { return Traj(a).to(1.0, b, Interp::Linear); }
    static Traj exp(double a, double b)    { return Traj(a).to(1.0, b, Interp::Exp); }
    static Traj smooth(double a, double b) { return Traj(a).to(1.0, b, Interp::Smooth); }
    /// Frequency sweep given in Hz, stored as MIDI notes (so a linear trajectory is an exponential sweep in Hz).
    static Traj pitch_hz(double a_hz, double b_hz, Interp i = Interp::Linear) { return Traj(hz_to_midi(a_hz)).to(1.0, hz_to_midi(b_hz), i); }
    /// Attack to `peak` over the first `attack` fraction, then exponential decay to `floor` at the end.
    static Traj ar(double attack, double peak = 1.0, double floor = 0.001) { attack = clampd(attack, 0.0, 1.0); return Traj(0.0).to(attack, peak, Interp::Smooth).to(1.0, floor, Interp::Exp); }
    /// Exponential decay from 1 to `floor`.
    static Traj decay(double floor = 0.001) { return Traj(1.0).to(1.0, floor, Interp::Exp); }
    /// `n` equal steps from a to b (a staircase), including both end values.
    static Traj stairs(double a, double b, int n)
    /// Add a key at normalised time `t` (must not be earlier than the previous key).
    Traj& to(double t, double v, Interp how = Interp::Linear)
    double at(double u) const noexcept
    Traj scaled(double k) const { Traj t = *this; for (auto& key : t.k_) key.v *= k; return t; }
    Traj shifted(double d) const { Traj t = *this; for (auto& key : t.k_) key.v += d; return t; }
    double min_value() const { double m = k_[0].v; for (auto& k : k_) m = std::min(m, k.v); return m; }
    double max_value() const { double m = k_[0].v; for (auto& k : k_) m = std::max(m, k.v); return m; }
    size_t key_count() const noexcept { return k_.size(); }

wauvio::sfx::struct Mod

/// Low-frequency modulator, evaluated statelessly from time (so any layer can be rendered in any order).
struct Mod
    enum class Shape { Sine, Triangle, Square, Saw, Smooth, Steps };   // Smooth/Steps = random (seeded)
    Shape  shape = Shape::Sine;
    double rate_hz = 0.0;   ///< cycles per second (ignored if `beats` > 0)
    double beats   = 0.0;   ///< if > 0: one cycle per this many beats (tempo-synced)
    double depth   = 0.0;   ///< output range is [-depth, +depth]
    double phase   = 0.0;   ///< cycles
    bool active() const noexcept { return depth != 0.0 && (rate_hz > 0.0 || beats > 0.0); }
    double eval(double t_sec, double bpm, uint64_t seed) const noexcept

wauvio::sfx::struct Source

struct Source
    SourceKind kind = SourceKind::None;
    Traj pitch = 60.0;                       ///< MIDI note (fractional allowed); used by every pitched kind
    // Osc
    Wave wave = Wave::Saw; int voices = 1; Traj detune_cents = 0.0; Traj pulse_width = 0.5;
    double sub = 0.0;                        ///< sine one octave below, relative level
    std::vector<double> ratios;              ///< non-empty: a stack of oscillators at pitch * ratio (e.g. metallic square stacks)
    Traj fm_index = 0.0; double fm_ratio = 1.0;   ///< phase modulation of the carrier by a sine at pitch * fm_ratio
    // Noise
    NoiseColor color = NoiseColor::White; Traj density = 200.0;   ///< density = events per second (Dust only)
    // Additive: each note (semitone offset from `pitch`) gets `harmonics` partials
    std::vector<double> notes { 0.0 }; int harmonics = 1; Traj rolloff = 1.0;   ///< partial n has amplitude n^-rolloff
    double inharmonicity = 0.0;              ///< partial n sits at n*sqrt(1 + B*n^2)
    Traj partial_spread = 0.0;               ///< random per-partial detune in cents (smooth, seeded)
    // Modal / Pluck: events happen at normalised times `hits` (+/- jitter); hit k plays pitch + hit_notes[k % size]
    std::vector<Mode> modes; std::vector<double> hits; std::vector<double> hit_notes; double hit_jitter = 0.0; double velocity_var = 0.3;
    double strike_ms = 1.5;                  ///< noise burst at each hit (the "click" of the attack)
    double t60_scale = 1.0;                  ///< multiplies every mode / string decay time
    double damping = 0.5;                    ///< Pluck: 0 bright & ringing .. 1 dull & short
    // Formant
    Traj vowel = 0.0;                        ///< 0=a 1=e 2=i 3=o 4=u, interpolated
    Traj formant_shift = 0.0;                ///< semitones, moves all formants
    double breath = 0.0;                     ///< 0 voiced .. 1 whispered (noise excitation)
    bool   noise_excited = false;
    // Grain
    GrainSource grain_source = GrainSource::Noise; Traj grain_ms = 60.0; Traj grain_pitch = 0.0;   ///< semitones
    double pitch_scatter = 0.0; Traj position = 0.0; double position_scatter = 0.2; double grain_chord = 0.0;
    // Chaos
    ChaosKind chaos = ChaosKind::Lorenz; Traj chaos_rate = 40.0;   ///< attractor evolution speed
    Traj chaos_depth = 12.0;                 ///< semitones of pitch modulation (carrier mode)
    double chaos_amp = 0.0;                  ///< amount the attractor also modulates the level
    bool   chaos_direct = false;             ///< output the attractor itself instead of an oscillator
    Traj logistic_r = 3.9;

wauvio::sfx::struct LayerSpec

struct LayerSpec
    std::string name;
    Source src;
    std::vector<Proc> chain;
    Traj env = 1.0;            ///< amplitude envelope applied to the source, before the chain (tails can ring on)
    Traj post = 1.0;           ///< amplitude automation applied after the chain
    double level = 1.0;        ///< linear peak level in the mix
    double start = 0.0;        ///< seconds from the start of the effect
    double length = -1.0;      ///< seconds of source; < 0 = until the end of the effect
    double tail = 0.0;         ///< extra seconds the chain may ring after the source ends
    double pan = 0.0;          ///< -1 left .. +1 right (equal power)
    double width = 0.0;        ///< 0 mono (just panned) .. 1 independently rendered L/R (decorrelated)
    double fade_in_ms = 0.5, fade_out_ms = 4.0;
    uint64_t seed_offset = 0;  ///< change to get a different random take of just this layer
    // sources
    LayerSpec& osc(Wave w, Traj pitch_midi) { src = Source(); src.kind = SourceKind::Osc; src.wave = w; src.pitch = pitch_midi; return *this; }
    LayerSpec& noise(NoiseColor c, Traj density = 200.0) { src = Source(); src.kind = SourceKind::Noise; src.color = c; src.density = density; return *this; }
    LayerSpec& additive(Traj pitch_midi, std::vector<double> notes, int harmonics, Traj rolloff = 1.0) { src = Source(); src.kind = SourceKind::Additive; src.pitch = pitch_midi; src.notes = std::move(notes); src.harmonics = harmonics; src.rolloff = rolloff; return *this; }
    LayerSpec& modal(std::vector<Mode> modes, Traj pitch_midi, std::vector<double> hits) { src = Source(); src.kind = SourceKind::Modal; src.modes = std::move(modes); src.pitch = pitch_midi; src.hits = std::move(hits); return *this; }
    LayerSpec& formant(Traj vowel, Traj pitch_midi, double breath = 0.0) { src = Source(); src.kind = SourceKind::Formant; src.vowel = vowel; src.pitch = pitch_midi; src.breath = breath; return *this; }
    LayerSpec& grain(GrainSource gs, Traj density, Traj size_ms, Traj pitch_midi = 60.0) { src = Source(); src.kind = SourceKind::Grain; src.grain_source = gs; src.density = density; src.grain_ms = size_ms; src.pitch = pitch_midi; return *this; }
    LayerSpec& pluck(Traj pitch_midi, std::vector<double> hits, double damping = 0.5) { src = Source(); src.kind = SourceKind::Pluck; src.pitch = pitch_midi; src.hits = std::move(hits); src.damping = damping; return *this; }
    LayerSpec& chaos(ChaosKind k, Traj rate, Traj pitch_midi = 60.0) { src = Source(); src.kind = SourceKind::Chaos; src.chaos = k; src.chaos_rate = rate; src.pitch = pitch_midi; return *this; }
    // source options
    LayerSpec& unison(int voices, Traj cents) { src.voices = voices; src.detune_cents = cents; return *this; }
    LayerSpec& fm(Traj index, double ratio) { src.fm_index = index; src.fm_ratio = ratio; return *this; }
    LayerSpec& sub(double g) { src.sub = g; return *this; }
    LayerSpec& stack(std::vector<double> ratios) { src.ratios = std::move(ratios); return *this; }
    LayerSpec& pw(Traj w) { src.pulse_width = w; return *this; }
    LayerSpec& inharmonic(double b) { src.inharmonicity = b; return *this; }
    LayerSpec& spread(Traj cents) { src.partial_spread = cents; return *this; }
    LayerSpec& hit_notes(std::vector<double> n) { src.hit_notes = std::move(n); return *this; }
    LayerSpec& hit_jitter(double j, double vel_var = 0.3) { src.hit_jitter = j; src.velocity_var = vel_var; return *this; }
    LayerSpec& strike(double ms) { src.strike_ms = ms; return *this; }
    LayerSpec& decay_scale(double s) { src.t60_scale = s; return *this; }
    LayerSpec& scatter(double pitch_semis, double position_frac) { src.pitch_scatter = pitch_semis; src.position_scatter = position_frac; return *this; }
    LayerSpec& grain_pos(Traj p) { src.position = p; return *this; }
    LayerSpec& grain_pitch(Traj semis) { src.grain_pitch = semis; return *this; }
    LayerSpec& shift(Traj semis) { src.formant_shift = semis; return *this; }
