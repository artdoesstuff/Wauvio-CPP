FX suite (wauvio/fx)

Modulation, delay and stereo

Modulation (effects/modulation.hpp)

  • FracDelay — ring-buffer delay with Catmull-Rom fractional reads: init(max_samples), push, read(d) (d ≥ 1), at(back), clear, capacity.
  • Tremolo — rate_hz 5, optional tempo sync (sync_bpm, sync_beats), depth 0.5, stereo_phase_deg (180 = alternating L/R), shape.
  • AutoPan — rate_hz 0.5, sync_bpm, sync_beats 4, depth 1, shape, phase_deg.
  • StereoChorus — rate_hz 0.6, depth_ms 2.5, delay_ms 14, voices 1–6 per channel (3), spread, mix 0.5, shape.
  • DualMono<T> — adapter that runs a core mono effect (anything with init(sr) and process(Buffer&), such as fx::basic::Flanger or Phaser) independently on each channel. Access sides with left()/right() or set both via configure(fn); set extra_tail_seconds if the wrapped effect rings.

Stereo (effects/stereo.hpp)

  • StereoImager — width (0 mono, 1 unchanged, 2 double), mid_gain_db, side_gain_db, mono_below_hz (4th-order high-pass on the side signal), balance.
  • Utility — gain_db, pan (equal power), mute, invert_left, invert_right, swap_channels, mono, remove_dc.

wauvio::fx::class FracDelay

class FracDelay
public:
    void init(size_t max_samples) { buf_.assign(max_samples + 8, 0.0f); pos_ = 0; }
    void clear() { std::fill(buf_.begin(), buf_.end(), 0.0f); pos_ = 0; }
    size_t capacity() const noexcept { return buf_.size() > 8 ? buf_.size() - 8 : 0; }
    void push(float x) noexcept { buf_[pos_] = x; pos_ = (pos_ + 1) % buf_.size(); }
    /// Read `d` samples back (d >= 1; 1 = the most recent push). Fractional delays use
    /// Catmull-Rom interpolation.
    float read(double d) const noexcept

wauvio::fx::class DualMono

class DualMono : public Effect
public:
    DualMono() = default;
    std::string name() const override { return "DualMono"; }
    T& left()  { return l_; }
    T& right() { return r_; }
    template <class F> DualMono& configure(F&& f) { f(l_); f(r_); return *this; }
    double extra_tail_seconds = 0.0;   ///< set if the wrapped effect rings (e.g. a flanger with feedback)
    double tail_seconds() const override { return extra_tail_seconds; }
    void reset() override { l_.init(sr_); r_.init(sr_); }
    using Effect::process;
    void process(float* L, float* R, size_t n) override

wauvio/fx/effects/delay.hpp

wauvio::fx::class StereoDelay

class StereoDelay : public Effect
public:
    // Time: either milliseconds or tempo sync (sync_bpm > 0 -> sync_beats_* are used).
    double time_ms        = 350.0;   ///< left (or both) delay time
    double time_ms_right  = -1.0;    ///< < 0 = same as left
    double sync_bpm       = 0.0;
    double sync_beats_left  = div::dotted(div::eighth);   ///< default: dotted eighth
    double sync_beats_right = -1.0;                       ///< < 0 = same as left
    double feedback       = 0.4;     ///< 0..0.98
    bool   ping_pong      = false;   ///< input into the left line, echoes bounce L -> R -> L ...
    double lowcut_hz      = 0.0;     ///< high-pass in the feedback path (0 = off)
    double highcut_hz     = 6000.0;  ///< low-pass in the feedback path (0 = off) - each repeat gets darker
    double mod_depth_ms   = 0.0;     ///< tape-style wobble
    double mod_rate_hz    = 0.4;
    double mix            = 0.35;
    std::string name() const override { return "StereoDelay"; }
    double tail_seconds() const override
    void reset() override { pos_ = 0; dl_[0].clear(); dl_[1].clear(); for (auto& f : lc_) f.reset(); for (auto& f : hc_) f.reset(); }
    using Effect::process;
    void process(float* L, float* R, size_t n) override
